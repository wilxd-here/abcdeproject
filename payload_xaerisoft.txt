 const app=A.initializeApp(FIREBASE_CONFIG),auth=Au.getAuth(app),db=Fs.getFirestore(app),col=Fs.collection(db,"messages"),P={},Q=new Set(),ud=i=>Fs.doc(db,"users",i);
 Au.getRedirectResult(auth).catch(e=>{console.error("Auth redirect error:",e)});
 Object.assign(XA,{ready:true,
  social:async p=>{
   const pr=p==="github"?new Au.GithubAuthProvider():new Au.GoogleAuthProvider();
   const cu=auth.currentUser;
   if(cu&&cu.isAnonymous)return Au.linkWithRedirect(cu,pr);
   return Au.signInWithRedirect(auth,pr);
  },
  signin:(e,p)=>Au.signInWithEmailAndPassword(auth,e,p),
  signup:async(n,e,p)=>{const cu=auth.currentUser;let c;if(cu&&cu.isAnonymous)c=await Au.linkWithCredential(cu,Au.EmailAuthProvider.credential(e,p));else c=await Au.createUserWithEmailAndPassword(auth,e,p);await Au.updateProfile(c.user,{displayName:n});XA.user=auth.currentUser;fire()},
  out:()=>Au.signOut(auth),
  rename:async n=>{await Au.updateProfile(auth.currentUser,{displayName:n});try{await Fs.setDoc(ud(auth.currentUser.uid),{name:n},{merge:true})}catch{}XA.user=auth.currentUser;fire()},
  guest:async n=>{const c=await Au.signInAnonymously(auth);await Au.updateProfile(c.user,{displayName:n});XA.user=auth.currentUser;fire()},
  setPhoto:async d=>{const cu=auth.currentUser;await Fs.setDoc(ud(cu.uid),{photo:d,name:cu.displayName||"",at:Fs.serverTimestamp()},{merge:true});P[cu.uid]={...P[cu.uid],photo:d};fire()},
  photo:id=>{const p=P[id];if(p===undefined&&!Q.has(id)){Q.add(id);Fs.getDoc(ud(id)).then(x=>{P[id]=x.exists()?x.data():{}}).catch(()=>{P[id]={}}).then(()=>fire())}return p&&p.photo||""},
  send:(t,o)=>{const u=auth.currentUser,d={text:t,uid:u.uid,name:u.displayName||(u.isAnonymous?"Tamu":"Pengguna"),photo:u.photoURL||"",at:Fs.serverTimestamp()},r=o&&o.replyTo,f=o&&o.fwd;
   if(r&&typeof r.id==="string"&&r.id)d.replyTo={id:r.id,uid:String(r.uid||""),name:String(r.name||"").slice(0,40),text:String(r.text||"").slice(0,120)};
   if(f)d.fwd={name:String(f.name||"").slice(0,40),uid:String(f.uid||"")};
   return Fs.addDoc(col,d)},
  react:(id,k)=>Fs.updateDoc(Fs.doc(db,"messages",id),{["reactions."+auth.currentUser.uid]:k}),
  unreact:id=>Fs.updateDoc(Fs.doc(db,"messages",id),{["reactions."+auth.currentUser.uid]:Fs.deleteField()}),
  del:id=>Fs.deleteDoc(Fs.doc(db,"messages",id)),
  report:(m,why)=>{const u=auth.currentUser;return Fs.setDoc(Fs.doc(db,"reports",m.id+"_"+u.uid),{msgId:m.id,msgUid:String(m.uid||""),msgText:String(m.text||"").slice(0,200),reason:why,reporterUid:u.uid,at:Fs.serverTimestamp()})},
  role:id=>{if(!id)return roleOf();XA.photo(id);return roleOf(id,P[id])},
  listen:(cb,er)=>Fs.onSnapshot(Fs.query(col,Fs.orderBy("at","desc"),Fs.limit(80)),s=>cb(s.docs.map(d=>({id:d.id,...d.data()})).reverse()),er)});
 Au.onAuthStateChanged(auth,u=>{XA.user=u;XA.ar=true;if(u)XA.photo(u.uid);fire()});
}catch(e){console.error(e);XA.error=e.message}}
XA.loaded=true;fire();
</script>
