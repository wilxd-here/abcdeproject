function ttParse(raw){
 let s=String(raw||"").trim();
 if(!s)return{err:"Tempel link video TikTok terlebih dahulu."};
 if(/\s/.test(s)){                              // teks share: ambil token yang berisi link TikTok
  const tok=s.split(/\s+/).filter(x=>/tiktok\.com\//i.test(x));
  if(tok.length!==1)return{err:"Link tidak valid. Tempel satu link TikTok saja."};
  s=tok[0].replace(/^["'<(]+|["'>).,]+$/g,"");
 }
 if(!/^https?:\/\//i.test(s))s="https://"+s;
 let u;try{u=new URL(s)}catch{return{err:"Format link tidak valid."}}
 if(u.protocol!=="https:"&&u.protocol!=="http:")return{err:"Link harus diawali http atau https."};
 if(!TT_HOSTS.test(u.hostname))return{err:"Link bukan dari TikTok. Gunakan link dari tiktok.com, vm.tiktok.com, atau vt.tiktok.com."};
 if(u.pathname.length<3)return{err:"Link belum mengarah ke video TikTok tertentu."};
 u.protocol="https:";
 return{url:u.toString()};
}

const isUrl=v=>typeof v==="string"&&/^https?:\/\/[^\s]+$/i.test(v.trim());
function walk(o,path,out,d){
 if(o==null||d>7)return;
 if(typeof o==="string"){if(isUrl(o))out.push({path:path.join(".").toLowerCase(),url:o.trim()});return}
 if(Array.isArray(o)){o.forEach((x,i)=>walk(x,path.concat(String(i)),out,d+1));return}
 if(typeof o==="object")for(const k in o)walk(o[k],path.concat(k),out,d+1);
}
function findKey(o,keys,test){
 const q=[o];let n=0;
 while(q.length&&n++<400){
  const c=q.shift();if(c==null||typeof c!=="object")continue;
  for(const k of keys){
   const hit=Object.keys(c).find(x=>x.toLowerCase()===k);
   if(hit!==undefined){const v=c[hit];if(!test||test(v))return v}
  }
  for(const k in c)if(c[k]&&typeof c[k]==="object")q.push(c[k]);
 }
 return undefined;
}
